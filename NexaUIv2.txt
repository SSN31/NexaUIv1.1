--[[
    NexaUI v2
    Sidebar + pages. Searchable selector instead of dropdown columns.
    Drop-in compatible with v1 scripts: same CreateWindow/CreateTab/CreateSection/
    CreateDivider/CreateToggle/CreateButton/CreateSlider/CreateDropdown/CreateLabel/
    CreateInput/CreateParagraph, same Window.Modules Get/Set, same
    NexaUI:Notify/SaveConfig/LoadConfig/Destroy/Toggle.
    Usage:
        local NexaUI = loadstring(readfile("path/to/NexaUIv2.txt"))()
        local Window = NexaUI:CreateWindow({
            Name = "My Script",
            Theme = {
                Accent = Color3.fromRGB(180, 100, 200),
                Font = Enum.Font.Gotham,
                BreathingEffect = true,
                NotificationPosition = "Top-Right",
                UITransparency = 0,
                UIScale = 100,
            }
        })
        local Tab = Window:CreateTab({Name = "Main", Icon = "◆"})
        Tab:CreateSection("SECTION NAME")
        Tab:CreateToggle({Name = "My Toggle", Default = false, Callback = function(v) end})
        Tab:CreateSlider({Name = "My Slider", Min = 0, Max = 100, Default = 50, Callback = function(v) end})
        Tab:CreateButton({Name = "My Button", Callback = function() end})
        Tab:CreateDropdown({Name = "My Pick", Options = {"A","B"}, Default = "A", Callback = function(v) end})
        Tab:CreateLabel({Text = "Hello"})
        Tab:CreateInput({Name = "My Input", PlaceholderText = "Type here", Callback = function(t) end})
        Tab:CreateParagraph({Title = "T", Content = "C"})
        NexaUI:SaveConfig("ConfigName")
        NexaUI:LoadConfig("ConfigName")
        NexaUI:Notify({Title = "Title", Content = "Content", Duration = 5})
        NexaUI:Destroy()
--]]

local KeyNameToCode = {}
for _, kc in ipairs(Enum.KeyCode:GetEnumItems()) do
    KeyNameToCode[kc.Name] = kc
end

local function Darken(c, f) return Color3.new(c.R*f, c.G*f, c.B*f) end
local function Lighten(c, f)
    return Color3.new(
        math.min(c.R+(1-c.R)*f,1), math.min(c.G+(1-c.G)*f,1), math.min(c.B+(1-c.B)*f,1))
end
local function EnabledBg(accent)
    local b = Color3.fromRGB(20,20,24)
    return Color3.new(b.R*0.8+accent.R*0.2, b.G*0.8+accent.G*0.2, b.B*0.8+accent.B*0.2)
end
local function Cr(p, r) local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0,r or 4); c.Parent = p; return c end

    function W:OpenSelector(mod)
        local sel = W._selector
        sel.openFor = mod
        sel.title.Text = mod._selectTitle or "Select"
        sel.search.Text = ""
        for _,b in ipairs(sel.optionBtns) do b:Destroy() end

        for _,opt in ipairs(mod._selectOptions or {}) do
            local ob = Instance.new("TextButton")
            ob.Size = UDim2.new(1, -4, 0, rowH)
            ob.BackgroundColor3 = (opt == mod:Get()) and EnabledBg(W.CurrentAccent) or ROW_BG
            ob.Text = "  " .. tostring(opt)
            ob.TextColor3 = (opt == mod:Get()) and Lighten(W.CurrentAccent, 0.4) or Color3.fromRGB(200,200,200)
            ob.Font = W.CurrentFont; ob.TextSize = 12
            ob.TextXAlignment = Enum.TextXAlignment.Left
            ob.BorderSizePixel = 0; ob.AutoButtonColor = false
            ob.ZIndex = 603; ob.Parent = sel.scroll; ob.LayoutOrder = #sel.scroll:GetChildren()
            Cr(ob, 4)
            table.insert(W._fonted, ob)
            table.insert(sel.optionBtns, ob)
            ob.MouseButton1Click:Connect(function()
                mod:Set(opt)
                if mod._selectCb then task.spawn(mod._selectCb, opt) end
                closeSelector()
            end)
        end
        selCatcher.Visible = true
        selPanel.Visible = true
        task.defer(function() selSearch:CaptureFocus() end)
    end

    function W:ApplyUITransparency(t)
        W.UITransparencyVal = (t or 0) * 100
        if W.UIOpen then W:RefreshFade() end
    end

    function W:RefreshFade()
        local ft = W.UITransparencyVal / 100
        if W.UIOpen then
            TS:Create(W.Backdrop, TweenInfo.new(0.25), {BackgroundTransparency = 0.45 + ft * 0.4}):Play()
        end
    end

    function W:StartBreathing()
        if W._breathConn then pcall(function() W._breathConn:Disconnect() end) end
        W._breathConn = RS.Heartbeat:Connect(function()
            local s = (math.sin(tick() * 2) + 1) / 2
            local c = W.CurrentAccent
            pcall(function()
                W._accentStroke.Color = Color3.new(
                    math.min(c.R + (1-c.R)*s*0.4, 1),
                    math.min(c.G + (1-c.G)*s*0.4, 1),
                    math.min(c.B + (1-c.B)*s*0.4, 1))
            end)
        end)
    end

    function W:StopBreathing()
        if W._breathConn then pcall(function() W._breathConn:Disconnect() end) end
        W._breathConn = nil
        pcall(function() W._accentStroke.Color = W.CurrentAccent end)
    end

    -- Open / close
    function W:ToggleUI()
        if W.ToggleAnimating then return end
        W.ToggleAnimating = true
        W.UIOpen = not W.UIOpen
        if W.UIOpen then
            win.Visible = true
            W.ScreenGui.Enabled = true
            W.Backdrop.Visible = true
            openBtn.Visible = false
            TS:Create(W.Backdrop, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {BackgroundTransparency = 0.55}):Play()
            task.delay(0.35, function() W.ToggleAnimating = false end)
        else
            TS:Create(W.Backdrop, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {BackgroundTransparency = 1}):Play()
            task.delay(0.32, function()
                if not W.UIOpen then
                    win.Visible = false
                    W.Backdrop.Visible = false
                    openBtn.Visible = true
                    W.ToggleAnimating = false
                end
            end)
        end
    end

    -- Search filter
    local function applySearch()
        local q = searchBox.Text:lower()
        for _,entry in ipairs(W._searchIdx) do
            entry.row.Visible = (q == "") or (entry.name:lower():find(q, 1, true) ~= nil)
        end
        for i,page in ipairs(W._pages) do
            local has = false
            if q ~= "" then
                for _,entry in ipairs(W._searchIdx) do
                    if entry.page == i and entry.row.Visible then has = true break end
                end
            end
            if page.dot then page.dot.Visible = has and (i ~= W._activePage) end
        end
    end
    searchBox:GetPropertyChangedSignal("Text"):Connect(applySearch)
    W._applySearch = applySearch

    function W:_registerSearch(row, name, pageIdx)
        table.insert(W._searchIdx, {row = row, name = tostring(name), page = pageIdx})
    end

    -- Page switching
    W._pageBtns = {}
    function W:showPage(idx)
        W._activePage = idx
        for i,page in ipairs(W._pages) do
            page.frame.Visible = (i == idx)

                btn._active = (i == idx)

                    btn._label.TextColor = Lighten(W.CurrentAccent, 0.4)

                    btn._frame.BackgroundColor3 = Color3.fromRGB(16,16,21)
                    btn._label.TextColor = TXT_DIM
                    btn._bar.Visible = false
                end
            end
        end
        W:_applySearch()
    end

            track.BorderSizePixel = 0; track.ZIndex = 12; track.Parent = row
            Cr(track, 3)
            local pct = (max > min) and ((val - min) / (max - min)) or 0
            local fill = Instance.new("Frame")
            fill.Size = UDim2.new(pct, 0, 1, 0)
            fill.BackgroundColor3 = self._w.CurrentAccent
            fill.BorderSizePixel = 0; fill.ZIndex = 12; fill.Parent = track
            Cr(fill, 3)
            local handle = Instance.new("TextButton")
            handle.Size = UDim2.new(0, 14, 0, 14)
            handle.Position = UDim2.new(pct, -7, 0.5, -7)
            handle.BackgroundColor3 = self._w.CurrentAccent
            handle.Text = ""; handle.BorderSizePixel = 0
            handle.AutoButtonColor = false; handle.ZIndex = 13; handle.Parent = track
            Cr(handle, 7)
            table.insert(self._w._updaters, function()
                fill.BackgroundColor3 = self._w.CurrentAccent
                handle.BackgroundColor3 = self._w.CurrentAccent
                vlbl.TextColor3 = self._w.CurrentAccent
            end)
            local sliding = false
            local dsx, dsp = 0, pct
            local function apply(nv, silent)
                val = math.floor(math.clamp(nv, min, max) + 0.5)
                pct = (max > min) and ((val - min) / (max - min)) or 0
                fill.Size = UDim2.new(pct, 0, 1, 0)
                handle.Position = UDim2.new(pct, -7, 0.5, -7)
                vlbl.Text = tostring(val)
                if cb and not silent then task.spawn(cb, val) end
            end

                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                    sliding = true; dsx = input.Position.X; dsp = pct
                end
            end)
            track.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                    local w = math.max(track.AbsoluteSize.X, 1)
                    local rel = math.clamp((input.Position.X - track.AbsolutePosition.X) / w, 0, 1)
                    apply(min + rel * (max - min))
                    sliding = true; dsx = input.Position.X; dsp = rel
                end
            end)
            local moveConn = UIS.InputChanged:Connect(function(input)
                if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch) then
                    local w = math.max(track.AbsoluteSize.X, 1)
                    apply(min + math.clamp(dsp + (input.Position.X - dsx) / w, 0, 1) * (max - min))
                end
            end)
            local endConn = UIS.InputEnded:Connect(function(input)
                if sliding and (input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch) then
                    sliding = false
                end
            end)
            table.insert(self._w._connections, moveConn)
            table.insert(self._w._connections, endConn)
            local so = {Frame = row, Get = function() return val end,
                Set = function(_, v, silent) apply(v, silent) end}
            self._w.Modules[opts.Name] = so
            return so
        end

    -- ============================================================
    -- CONFIG SYSTEM
    -- ============================================================
    function W:SetConfigName(name)
        self._configName = name or "Default"
    end

    function W:SaveConfig(name)
        name = name or self._configName
        if not isfolder(self.ConfigFolder) then makefolder(self.ConfigFolder) end
        local cfg = {}
        for mn,mod in pairs(self.Modules) do if mod.Get then cfg[mn] = mod:Get() end end
        cfg._AccentName = self.CurrentAccentName; cfg._FontName = self.CurrentFontName
        cfg._NotifPos = self.CurrentNotifPos; cfg._Breathing = self.BreathingEnabled
        cfg._UITransparency = self.UITransparencyVal; cfg._UIScale = self.UIScaleVal*100
        cfg._Keybinds = self.Keybinds
        writefile(self.ConfigFolder.."/"..name..".json", HS:JSONEncode(cfg))
        self:Notify({Title="Config",Content="Saved: "..name,Type="success",Duration=2})
    end

    function W:LoadConfig(name)
        name = name or self._configName
        local path = self.ConfigFolder.."/"..name..".json"
        if not isfile(path) then self:Notify({Title="Config",Content="Not found: "..name,Type="warning",Duration=2}) return end
        local ok, data = pcall(function() return HS:JSONDecode(readfile(path)) end)
        if not ok or type(data) ~= "table" then self:Notify({Title="Config",Content="Failed to load",Type="alert",Duration=2}) return end
        for mn,mod in pairs(self.Modules) do if data[mn] ~= nil and mod.Set then mod:Set(data[mn], true) end end
        if data._AccentName and AccentPresets[data._AccentName] then
            self.CurrentAccentName = data._AccentName; self.CurrentAccent = AccentPresets[self.CurrentAccentName]
            self:ApplyAccentColor(); if self.BreathingEnabled then self:StopBreathing() self:StartBreathing() end
        end
        if data._FontName and FontPresets[data._FontName] then self.CurrentFontName = data._FontName; self.CurrentFont = FontPresets[self.CurrentFontName]; self:ApplyFont() end
        if data._NotifPos then self.CurrentNotifPos = data._NotifPos end
        if data._Breathing ~= nil then self.BreathingEnabled = data._Breathing; if self.BreathingEnabled then self:StartBreathing() else self:StopBreathing() end end
        if data._UITransparency then self.UITransparencyVal = data._UITransparency; self:ApplyUITransparency(self.UITransparencyVal/100) end
        if data._UIScale then
            self.UIScaleVal = data._UIScale/100
            if self.WindowScale then self.WindowScale.Scale = self.UIScaleVal end
        end
        if data._Keybinds then self.Keybinds = data._Keybinds end
        self:Notify({Title="Config",Content="Loaded: "..name,Type="success",Duration=2})
    end

    function W:Destroy()
        self:StopBreathing()
        if W._toggleConn then pcall(function() W._toggleConn:Disconnect() end) end
        for _, conn in ipairs(W._connections) do pcall(function() conn:Disconnect() end) end
        W._connections = {}
        for _,f in pairs(self.ActiveNotifs) do if f then pcall(function() f:Destroy() end) end end
        self.ActiveNotifs = {}
        if self.ScreenGui then pcall(function() self.ScreenGui:Destroy() end) end
    end

    W._toggleConn = UIS.InputBegan:Connect(function(input, gpe)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            W:HideContextMenu()
        end
        if input.KeyCode == Enum.KeyCode.Escape and W._selector.openFor then
            W._closeSelector()
            return
        end
        if input.KeyCode == W.KeybindToggle or input.KeyCode == Enum.KeyCode.Insert then
            W:ToggleUI()
            return
        end
        if gpe then return end
        for mn, kn in pairs(W.Keybinds) do
            if input.KeyCode == Enum.KeyCode[kn] then
                local tm = W.Modules[mn]
                if tm and tm.Get and tm.Set then tm:Set(not tm:Get(), false) end
            end
        end
    end)
    table.insert(W._connections, W._toggleConn)
