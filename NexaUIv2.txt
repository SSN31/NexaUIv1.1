--[[
    NexaUI v2.3
    Sidebar + pages. Searchable selector instead of dropdown columns.
    Drop-in compatible with v1 scripts: same CreateWindow/CreateTab/CreateSection/
    CreateDivider/CreateToggle/CreateButton/CreateSlider/CreateDropdown/CreateLabel/
    CreateInput/CreateParagraph, same Window.Modules Get/Set, same
    NexaUI:Notify/SaveConfig/LoadConfig/Destroy/Toggle.
    v2.3 fixes (Oct 2026 audit 2, from Test Hub screenshot):
      - FIX sidebar tabs rendered BELOW the window. Root cause: the 1px sidebar
        edge divider shared the sidebar's UIListLayout, which lays out every
        child GuiObject. With SizeY scale 1 it ate the full height and pushed all
        buttons out. Sidebar now has no layout; buttons live in sideStack, edge
        is a manual sibling. Buttons parent sideStack with Size (1,0).
      - ADD load animation: window pops 0.9->1 (Back.Out) + backdrop flash,
        tab buttons stagger-fade in, cards already pop on creation, reopen pops.
      - ADD micro-animations: tab select/hover tweens, toggle knob/color tweens,
        button/dropdown hover tweens + chevron nudge, card hover edge glow,
        selector option hover.
      - FIX/ENHANCE breathing outlines: added outer halo stroke (Thickness 3)
        + pulse now lerps Color AND Transparency (±12%, ~4s period) on window
        edge, halo, open-button ring, selector, context menu, active tab bars.
        Old ±7% color-only pulse wasnear-invisible. Stop restores exact values.
    v2.2 fixes (Oct 2026 audit):
      - FIX crash: rowH() was nil (only a local number in OpenSelector).
        Added rowHeight() helper; Toggle/Button/Dropdown/Input now size correctly.
      - FIX LayoutOrder missing on Toggle/Button/Dropdown (all order 0).
      - FIX Theme tab stole focus (was page 1, shown first). Theme pinned last,
        first user tab auto-shown.
      - FIX HUD never visible + listed theme toggles. HUD visible, ignores
        Breathing/Accent/Font/Notifications/FOV/Transparency/Scale.
      - FIX re-execution leak: CreateWindow destroys previous window.
      - FIX context menu ate its own clicks (UIS InputBegan hid it before Click).
      - FIX centered notifs offset double (Anchor 0.5 + -W/2).
      - FIX notif key collision (tick()) -> monotonic sequence.
      - FIX accent change left dots/bars/HUD stale; breathing stop restores exact accent.
      - FIX slider/dropdown Set(..., silent) never re-applied on LoadConfig.
        Now always fires callback (silent only suppresses notify spam).
      - FIX FOV never restored; UI Scale now scales HUD too.
      - FIX config file funcs error outside executors (pcall + friendly notify).
      - FIX ViewportSize/resize never hooked (only CurrentCamera swap).
      - FIX card header/div LayoutOrder collision; Card:CreateSection support;
        Section/Divider/Label return self for card chaining.
    Usage:
        local NexaUI = loadstring(readfile("path/to/NexaUIv2.txt"))()
        local Window = NexaUI:CreateWindow({
            Name = "My Script",
            Theme = {
                Accent = Color3.fromRGB(180, 100, 200),
                Font = Enum.Font.Gotham,
                BreathingEffect = true,
                NotificationPosition = "Top-Right",
                UITransparency = 0,
                UIScale = 100,
            }
        })
        local Tab = Window:CreateTab({Name = "Main", Icon = "◆"})
        Tab:CreateSection("SECTION NAME")
        local Card = Tab:CreateCard("Farming", "🌾") -- 2-col cards, 1-col on mobile
        Tab:CreateToggle({Name = "My Toggle", Default = false, Callback = function(v) end})
        Tab:CreateSlider({Name = "My Slider", Min = 0, Max = 100, Default = 50, Callback = function(v) end})
        Tab:CreateButton({Name = "My Button", Callback = function() end})
        Tab:CreateDropdown({Name = "My Pick", Options = {"A","B"}, Default = "A", Callback = function(v) end})
        Tab:CreateLabel({Text = "Hello"})
        Tab:CreateInput({Name = "My Input", PlaceholderText = "Type here", Callback = function(t) end})
        Tab:CreateParagraph({Title = "T", Content = "C"})
        NexaUI:SaveConfig("ConfigName")
        NexaUI:LoadConfig("ConfigName")
        NexaUI:Notify({Title = "Title", Content = "Content", Duration = 5})
        NexaUI:Destroy()
--]]

local function Darken(c, f) return Color3.new(c.R*f, c.G*f, c.B*f) end
local function Lighten(c, f)
    return Color3.new(
        math.min(c.R+(1-c.R)*f,1), math.min(c.G+(1-c.G)*f,1), math.min(c.B+(1-c.B)*f,1))
end
local function EnabledBg(accent)
    local b = Color3.fromRGB(20,20,24)
    return Color3.new(b.R*0.8+accent.R*0.2, b.G*0.8+accent.G*0.2, b.B*0.8+accent.B*0.2)
end
local function Cr(p, r) local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0,r or 4); c.Parent = p; return c end

function NexaUI:CreateWindow(config)
    -- Re-execution safety: destroy previous window so old UI never lingers.
    -- Required for executor re-runs (test: "Re-execute script -> old UI must vanish").
    if NexaUI._lastWindow and NexaUI._lastWindow.ScreenGui then
        pcall(function() NexaUI._lastWindow:Destroy() end)
    end
    NexaUI._lastWindow = nil

    config = config or {}
    local theme = config.Theme or {}

    local W = setmetatable({}, {__index = NexaUI})

    W.CurrentAccent = theme.Accent or AccentPresets.Purple
    W.CurrentAccentName = "Purple"
    if typeof(theme.Accent) == "string" and AccentPresets[theme.Accent] then
        W.CurrentAccentName = theme.Accent
        W.CurrentAccent = AccentPresets[theme.Accent]
    end
    W.CurrentFont = theme.Font or Enum.Font.Gotham
    W.CurrentFontName = "Gotham"
    for n,f in pairs(FontPresets) do if f == W.CurrentFont then W.CurrentFontName = n break end end
    W.BreathingEnabled = theme.BreathingEffect or false
    W.CurrentNotifPos = theme.NotificationPosition or "Top-Right"
    W.UITransparencyVal = theme.UITransparency or 0
    W.UIScaleVal = (theme.UIScale or 100) / 100
    W.KeybindToggle = theme.Keybind or Enum.KeyCode.RightShift
    if typeof(W.KeybindToggle) == "string" then
        W.KeybindToggle = KeyNameToCode[W.KeybindToggle] or Enum.KeyCode.RightShift
    end
    W.UIOpen = true
    W.ToggleAnimating = false
    W.Compact = false
    W.Modules = {}
    W.UIStrokes = {}
    W.Keybinds = {}
    W.ActiveNotifs = {}
    W._notifOrder = {}
    W.ConfigFolder = "NexaHub_Configs"
    W._configName = "Default"
    W._connections = {}
    W._updaters = {}
    W._fonted = {}
    W._accentTexts = {}
    W._accentStrokes = {}
    W._breathTargets = {}
    W._surfaces = {}
    W._sectionTitles = {}
    W._pageBtns = {}
    W._listening = nil
    W._searchIdx = {}
    W._pages = {}
    W._activePage = 1
    W._breathConn = nil
    W._loading = false
    W._notifSeq = 0
    W._themeBtnRec = nil
    W.NotifColors = {
        info = {bg = Color3.fromRGB(18,18,26), bar = Color3.fromRGB(96,165,255)},
        success = {bg = Color3.fromRGB(16,24,18), bar = Color3.fromRGB(96,216,118)},
        warning = {bg = Color3.fromRGB(26,22,14), bar = Color3.fromRGB(255,176,32)},
        alert = {bg = Color3.fromRGB(28,16,16), bar = Color3.fromRGB(255,76,76)},
    }

    -- Surface + breathing registries. Defined early: window/sidebar/titlebar
    -- setup below calls them while CreateWindow is still executing.
    function W:_surface(obj, base)
        base = base or 0
        table.insert(W._surfaces, {obj = obj, base = base})
        pcall(function()
            obj.BackgroundTransparency = math.clamp(base + W.UITransparencyVal / 100, 0, 1)
        end)
    end

    function W:_registerBreath(obj, prop)
        table.insert(W._breathTargets, {obj = obj, prop = prop})
    end

    -- Row height helper (compact = touch-friendly). Used by Toggle/Button/
    -- Dropdown/Input builders. Must be defined before any builder runs.
    local function rowHeight()
        return (W.Compact and 34 or 28)
    end

    -- ScreenGui (single gethui call, safe fallback for non-executor / Studio)
    W.ScreenGui = Instance.new("ScreenGui")
    W.ScreenGui.Name = "NexaHub_Yurei"
    W.ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    W.ScreenGui.ResetOnSpawn = false
    W.ScreenGui.DisplayOrder = 999
    local secureParent = nil
    do
        local ok, res = pcall(function()
            local f = (typeof(gethui) == "function" and gethui)
                or (typeof(get_hidden_gui) == "function" and get_hidden_gui)
            if f then return f() end
            return nil
        end)
        if ok and typeof(res) == "Instance" then
            secureParent = res
        else
            secureParent = nil
            pcall(function() secureParent = game:GetService("CoreGui") end)
        end
    end
    if not secureParent then
        secureParent = plr and plr:FindFirstChildOfClass("PlayerGui")
        if not secureParent and plr then
            pcall(function() secureParent = plr:WaitForChild("PlayerGui", 5) end)
        end
    end
    W.ScreenGui.IgnoreGuiInset = true
    W.ScreenGui.Parent = secureParent

    -- Watermark
    local wm = Instance.new("TextLabel")
    wm.Name = "Watermark"; wm.AnchorPoint = Vector2.new(1,1)
    wm.Position = UDim2.new(1,-12,1,-12); wm.Size = UDim2.new(0,220,0,22)
    wm.BackgroundTransparency = 1
    wm.Text = "NexaUI v2 • " .. (config.Name or "Script")
    wm.TextColor3 = Color3.fromRGB(235,235,245); wm.TextTransparency = 0.45
    wm.Font = Enum.Font.GothamMedium; wm.TextSize = 14
    wm.TextXAlignment = Enum.TextXAlignment.Right
    wm.ZIndex = 1000; wm.Parent = W.ScreenGui
    local wms = Instance.new("UIStroke"); wms.Color = Color3.new(0,0,0); wms.Thickness = 1; wms.Transparency = 0.7; wms.Parent = wm

    W.Backdrop = Instance.new("Frame")
    W.Backdrop.Name = "Backdrop"
    W.Backdrop.Size = UDim2.new(1, 0, 1, 0)
    W.Backdrop.BackgroundColor3 = Color3.new(0,0,0)
    W.Backdrop.BackgroundTransparency = 0.55
    W.Backdrop.BorderSizePixel = 0
    W.Backdrop.ZIndex = 1
    W.Backdrop.Visible = false
    W.Backdrop.Parent = W.ScreenGui

    -- Main window

    win.Name = "Window"
    win.AnchorPoint = Vector2.new(0.5, 0.5)
    win.Position = UDim2.new(0.5, 0, 0.5, 0)
    win.Size = UDim2.new(0, 680, 0, 440)
    win.BackgroundColor3 = Color3.fromRGB(14,14,18)
    win.BorderSizePixel = 0
    win.ZIndex = 10
    win.Parent = W.ScreenGui
    Cr(win, 8)
    W.WindowFrame = win
    local winStroke = Instance.new("UIStroke", win)
    winStroke.Color = W.CurrentAccent; winStroke.Thickness = 1; winStroke.Transparency = 0.35
    W._accentStroke = winStroke
    W:_surface(win)
    W:_registerBreath(winStroke, "Color")
    -- Outer halo: makes the breathe/pulse actually visible (screenshot showed a
    -- flat static edge). Thickness 3, mostly transparent, pulsed harder.
    local winGlow = Instance.new("UIStroke", win)
    winGlow.Name = "Glow"
    winGlow.Color = W.CurrentAccent; winGlow.Thickness = 3; winGlow.Transparency = 0.85
    W._glowStroke = winGlow
    W:_registerBreath(winGlow, "Color")
    local winScale = Instance.new("UIScale", win)
    winScale.Scale = W.UIScaleVal
    W.WindowScale = winScale
    local winConstraint = Instance.new("UISizeConstraint", win)
    winConstraint.MinSize = Vector2.new(420, 300)

    -- Title bar
    local titlebar = Instance.new("Frame")
    titlebar.Name = "Titlebar"
    titlebar.Size = UDim2.new(1, 0, 0, 44)
    titlebar.BackgroundColor3 = Color3.fromRGB(18,18,24)
    titlebar.BorderSizePixel = 0
    titlebar.ZIndex = 11
    titlebar.Parent = win
    W:_surface(titlebar)
    local titleEdge = Instance.new("Frame")
    titleEdge.Size = UDim2.new(1, 0, 0, 1)
    titleEdge.Position = UDim2.new(0, 0, 1, -1)
    titleEdge.BackgroundColor3 = BORDER
    titleEdge.BorderSizePixel = 0
    titleEdge.ZIndex = 12
    titleEdge.Parent = titlebar
    local titleIcon = Instance.new("TextLabel")
    titleIcon.Size = UDim2.new(0, 34, 1, 0)
    titleIcon.BackgroundTransparency = 1
    titleIcon.Text = "❖"
    titleIcon.TextColor3 = W.CurrentAccent
    titleIcon.Font = Enum.Font.GothamBold; titleIcon.TextSize = 18
    titleIcon.ZIndex = 12; titleIcon.Parent = titlebar
    table.insert(W._fonted, titleIcon)
    W._titleIcon = titleIcon
    local titleLbl = Instance.new("TextLabel")
    titleLbl.Size = UDim2.new(0, 180, 1, 0)
    titleLbl.Position = UDim2.new(0, 34, 0, 0)
    titleLbl.BackgroundTransparency = 1
    titleLbl.Text = config.Name or "NexaUI"
    titleLbl.TextColor3 = Color3.fromRGB(235,235,245)
    titleLbl.Font = W.CurrentFont; titleLbl.TextSize = 14
    titleLbl.TextXAlignment = Enum.TextXAlignment.Left
    titleLbl.ZIndex = 12; titleLbl.Parent = titlebar
    table.insert(W._fonted, titleLbl)
    -- Search box
    local searchBox = Instance.new("TextBox")
    searchBox.Name = "Search"
    searchBox.Size = UDim2.new(1, -300, 0, 26)
    searchBox.Position = UDim2.new(0, 214, 0, 9)
    searchBox.BackgroundColor3 = Color3.fromRGB(22,22,28)
    searchBox.Text = ""; searchBox.PlaceholderText = "Search..."
    searchBox.PlaceholderColor3 = Color3.fromRGB(100,100,110)
    searchBox.TextColor3 = Color3.fromRGB(210,210,220)
    searchBox.Font = W.CurrentFont; searchBox.TextSize = 12
    searchBox.TextXAlignment = Enum.TextXAlignment.Left
    searchBox.ClearTextOnFocus = false
    searchBox.BorderSizePixel = 0; searchBox.ZIndex = 12; searchBox.Parent = titlebar
    Cr(searchBox, 5)
    local searchPad = Instance.new("UIPadding", searchBox)
    searchPad.PaddingLeft = UDim.new(0, 10); searchPad.PaddingRight = UDim.new(0, 10)
            table.insert(W._fonted, searchBox)
            W:_surface(searchBox)
    W._searchBox = searchBox
    local searchStroke = Instance.new("UIStroke", searchBox)
    searchStroke.Color = BORDER; searchStroke.Thickness = 1
    -- Toggle (hide) button
    local hideBtn = Instance.new("TextButton")
    hideBtn.Size = UDim2.new(0, 30, 0, 26)
    hideBtn.Position = UDim2.new(1, -40, 0, 9)
    hideBtn.BackgroundColor3 = Color3.fromRGB(22,22,28)
    hideBtn.Text = "–"; hideBtn.TextColor3 = TXT_DIM
    hideBtn.Font = Enum.Font.GothamBold; hideBtn.TextSize = 16
    hideBtn.BorderSizePixel = 0; hideBtn.ZIndex = 12; hideBtn.Parent = titlebar
    Cr(hideBtn, 5)
    hideBtn.MouseButton1Click:Connect(function() W:ToggleUI() end)

    -- Sidebar (NOTE: no UIListLayout directly on sidebar. The 1px edge divider
    -- must NOT share a layout container with the buttons: UIListLayout lays out
    -- every child GuiObject, so an edge with SizeY scale 1 would eat the full
    -- height and push all tab buttons below the window. Buttons live in sideStack.)
    local sidebar = Instance.new("Frame")
    sidebar.Name = "Sidebar"
    sidebar.Size = UDim2.new(0, 150, 1, -44)
    sidebar.Position = UDim2.new(0, 0, 0, 44)
    sidebar.BackgroundColor3 = Color3.fromRGB(16,16,21)
    sidebar.BorderSizePixel = 0
    sidebar.ZIndex = 11
    sidebar.Parent = win
    W.Sidebar = sidebar
    W:_surface(sidebar)

    sideEdge.Name = "Edge"
    sideEdge.Size = UDim2.new(0, 1, 1, 0)
    sideEdge.Position = UDim2.new(1, -1, 0, 0)
    sideEdge.BackgroundColor3 = BORDER
    sideEdge.BorderSizePixel = 0
    sideEdge.ZIndex = 12
    sideEdge.Parent = sidebar
    local sideStack = Instance.new("Frame")
    sideStack.Name = "Stack"
    sideStack.Size = UDim2.new(1, 0, 1, 0)
    sideStack.BackgroundTransparency = 1
    sideStack.BorderSizePixel = 0
    sideStack.ZIndex = 11
    sideStack.Parent = sidebar
    local sideList = Instance.new("UIListLayout", sideStack)
    sideList.SortOrder = Enum.SortOrder.LayoutOrder
    sideList.Padding = UDim.new(0, 4)
    local sidePad = Instance.new("UIPadding", sideStack)
    sidePad.PaddingTop = UDim.new(0, 8); sidePad.PaddingLeft = UDim.new(0, 8)
    sidePad.PaddingRight = UDim.new(0, 8)
    W.SideStack = sideStack

    -- Content holder
    local content = Instance.new("Frame")
    content.Name = "Content"
    content.Size = UDim2.new(1, -150, 1, -44)
    content.Position = UDim2.new(0, 150, 0, 44)
    content.BackgroundTransparency = 1
    content.BorderSizePixel = 0
    content.ZIndex = 11
    content.ClipsDescendants = true
    content.Parent = win
    W.Content = content

    -- Group transparency wrapper via CanvasGroup-like fade: tween a proxy value
    -- (kept simple: fade backdrop + window group transparency field)
    W._fade = 0

    -- Dragging (title bar, mouse + touch)
    local dragging, dragStart, startPos = false, nil, nil
    titlebar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = win.Position
        end
    end)
    local dragConn = UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            win.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    local dragEnd = UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    W._connections = {dragConn, dragEnd}

    -- Floating open button (mobile friendly)
    local openBtn = Instance.new("TextButton")
    openBtn.Name = "OpenButton"
    openBtn.Size = UDim2.new(0, 46, 0, 46)
    openBtn.Position = UDim2.new(1, -60, 1, -120)
    openBtn.BackgroundColor3 = Color3.fromRGB(18,18,24)
    openBtn.Text = "N"; openBtn.TextColor3 = W.CurrentAccent
    openBtn.Font = Enum.Font.GothamBold; openBtn.TextSize = 20
    openBtn.BorderSizePixel = 0; openBtn.ZIndex = 999
    openBtn.Visible = false
    openBtn.Parent = W.ScreenGui
    Cr(openBtn, 23)
    local openStroke = Instance.new("UIStroke", openBtn)
    openStroke.Color = W.CurrentAccent; openStroke.Thickness = 1.5
    W:_surface(openBtn)
    W:_registerBreath(openStroke, "Color")
    openBtn.MouseButton1Click:Connect(function() W:ToggleUI() end)
    openBtn.MouseEnter:Connect(function() openBtn.TextColor3 = Color3.fromRGB(255,255,255) end)
    openBtn.MouseLeave:Connect(function() openBtn.TextColor3 = W.CurrentAccent end)
    W.OpenButton = openBtn
    W._openStroke = openStroke
    -- drag the open button too
    local obDrag, obStart, obPos = false, nil, nil
    openBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            obDrag = true; obStart = input.Position; obPos = openBtn.Position
        end

    local obMove = UIS.InputChanged:Connect(function(input)
        if obDrag and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - obStart
            openBtn.Position = UDim2.new(obPos.X.Scale, obPos.X.Offset + d.X, obPos.Y.Scale, obPos.Y.Offset + d.Y)
        end
    end)
    local obEnd = UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            if obDrag then obDrag = false end
        end
    end)
    table.insert(W._connections, obMove)
    table.insert(W._connections, obEnd)

    -- Context menu (per-module keybinds)
    W.ContextMenu = Instance.new("Frame")
    W.ContextMenu.Size = UDim2.new(0, 170, 0, 0)
    W.ContextMenu.AutomaticSize = Enum.AutomaticSize.Y
    W.ContextMenu.BackgroundColor3 = Color3.fromRGB(16, 16, 24)

    W.ContextMenu.ZIndex = 700
    W.ContextMenu.Parent = W.ScreenGui
    Cr(W.ContextMenu, 6)
    local cmStroke = Instance.new("UIStroke", W.ContextMenu)
    cmStroke.Color = W.CurrentAccent
    W:_registerBreath(cmStroke, "Color")
    W:_surface(W.ContextMenu)
    Instance.new("UIListLayout", W.ContextMenu).Padding = UDim.new(0, 2)
    local cmPad = Instance.new("UIPadding", W.ContextMenu)
    cmPad.PaddingTop = UDim.new(0,4); cmPad.PaddingBottom = UDim.new(0,4)
    cmPad.PaddingLeft = UDim.new(0,4); cmPad.PaddingRight = UDim.new(0,4)

    function W:_pointInGui(gui, pt)
        local ok, res = pcall(function()
            local pos = gui.AbsolutePosition
            local size = gui.AbsoluteSize
            return pt.X >= pos.X and pt.X <= pos.X + size.X
                and pt.Y >= pos.Y and pt.Y <= pos.Y + size.Y
        end)
        return ok and res or false
    end

    function W:HideContextMenu()
        if not W.ContextMenu.Visible then return end
        W.ContextMenu.Visible = false
        for _,c in pairs(W.ContextMenu:GetChildren()) do
            if c:IsA("TextButton") then c:Destroy() end
        end
    end

    function W:ShowContextMenu(options, position)
        W:HideContextMenu()
        for _,opt in ipairs(options) do
            if opt.Separator then
                local sep = Instance.new("Frame")
                sep.Size = UDim2.new(1, 0, 0, 1)
                sep.BackgroundColor3 = BORDER
                sep.BorderSizePixel = 0
                sep.ZIndex = 701
                sep.Parent = W.ContextMenu
            elseif opt.Header then
                local h = Instance.new("TextLabel")
                h.Size = UDim2.new(1, 0, 0, 22)
                h.BackgroundTransparency = 1

    -- Breathing/pulse outlines: one Heartbeat lerps every registered accent
    -- stroke/bar in Color AND Transparency, so the window edge + active tab bar +
    -- open-button ring visibly glow on a ~4s period. Single connection, no tweens.
    function W:StartBreathing()
        W:StopBreathing(true)
        local bases = {}
        for _, t in ipairs(W._breathTargets) do
            if t.obj and t.obj.Parent then
                pcall(function()
                    bases[t] = {
                        color = t.obj[t.prop],
                        transp = (typeof(t.obj.Transparency) == "number") and t.obj.Transparency or nil,
                        glow = (t.obj == W._glowStroke),
                    }
                end)
            end
        end
        W._breathConn = RS.Heartbeat:Connect(function()
            local s = math.sin(tick() * 1.6)
            local f = 1 + 0.12 * s
            for t, base in pairs(bases) do
                if typeof(base.color) == "Color3" then
                    pcall(function()
                        if t.obj and t.obj.Parent then
                            t.obj[t.prop] = Color3.new(
                                math.clamp(base.color.R * f, 0, 1),
                                math.clamp(base.color.G * f, 0, 1),
                                math.clamp(base.color.B * f, 0, 1))
                            -- Glow halo pulses harder than the hairline edge.
                            if base.transp ~= nil and t.obj:IsA("UIStroke") then
                                local amp = base.glow and 0.18 or 0.1
                                t.obj.Transparency = math.clamp(base.transp - amp * s, 0, 1)
                            end
                        end
                    end)
                end
            end
        end)
    end

    function W:StopBreathing(silent)
        if W._breathConn then pcall(function() W._breathConn:Disconnect() end) end
        W._breathConn = nil
        -- Restore exact accent (breathing leaves targets slightly tinted).
        for _, t in ipairs(W._breathTargets or {}) do
            if t.obj and t.obj.Parent then
                pcall(function()
                    if t.prop == "Color" or t.prop == "BackgroundColor3" then
                        t.obj[t.prop] = W.CurrentAccent
                    end
                end)
            end
        end
        if W._glowStroke and W._glowStroke.Parent then
            pcall(function()
                W._glowStroke.Color = W.CurrentAccent
                W._glowStroke.Transparency = 0.85
            end)
        end
        pcall(function()
            if winStroke and winStroke.Parent then winStroke.Transparency = 0.35 end
        end)
        -- Sidebar active bar may belong to an inactive button; re-assert it.
        for _,b in ipairs(W._pageBtns or {}) do
            if b._active and b._bar and b._bar.Parent then
                pcall(function() b._bar.BackgroundColor3 = W.CurrentAccent end)
            end
        end
        if not silent then
            -- Keep visuals snapped when breathing is toggled off at runtime.
            pcall(function() W:applyTransparency() end)
        end
    end

    function W:ToggleUI()
        if W.ToggleAnimating then return end
        W.ToggleAnimating = true
        W.UIOpen = not W.UIOpen

            task.delay(0.32, function()
                if not W.UIOpen then

                    openBtn.Visible = true
                    W.ToggleAnimating = false
                end
            end)
        end
    end

    -- Search filter (also collapses empty CardRows so gaps don't linger)
    local function applySearch()
        local q = searchBox.Text:lower()
        for _,entry in ipairs(W._searchIdx) do
            if entry.row and entry.row.Parent then
                entry.row.Visible = (q == "") or (entry.name:lower():find(q, 1, true) ~= nil)
            end
        end
        -- Hide CardRow frames whose cards are all hidden.
        for _,page in ipairs(W._pages) do
            if page.cardRows then
                for _,crow in ipairs(page.cardRows) do
                    if crow and crow.Parent then
                        local any = (q == "")
                        if not any then
                            for _,ch in ipairs(crow:GetChildren()) do
                                if ch:IsA("GuiObject") and ch.Visible then any = true break end
                            end
                        end
                        crow.Visible = any
                    end
                end
            end
        end
        for i,page in ipairs(W._pages) do
            local has = false
            if q ~= "" then
                for _,entry in ipairs(W._searchIdx) do
                    if entry.page == i and entry.row and entry.row.Parent and entry.row.Visible then has = true break end
                end
            end
            if page.dot then page.dot.Visible = has and (i ~= W._activePage) end
        end
    end

                    btn._frame.BackgroundColor3 = EnabledBg(W.CurrentAccent)
                    btn._label.TextColor3 = Lighten(W.CurrentAccent, 0.4)
                else
                    pcall(function()
                        TS:Create(btn._frame, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),

            end

    end

    local function hookResponsive()

    end

            if cam and W._camSizeConn then
                pcall(function() W._camSizeConn:Disconnect() end)
                W._camSizeConn = nil
            end
            if cam then
                W._camSizeConn = cam:GetPropertyChangedSignal("ViewportSize"):Connect(function()
                    task.defer(hookResponsive)
                end)
                table.insert(W._connections, W._camSizeConn)
            end
        end)
    end)
    table.insert(W._connections, viewConn)
    -- Live resize (PC resizes, mobile rotation) - CurrentCamera swap alone misses these.

        local cam = Workspace.CurrentCamera
        if cam then
            W._camSizeConn = cam:GetPropertyChangedSignal("ViewportSize"):Connect(function()
                task.defer(hookResponsive)
            end)

        end
    end)

    function W:CreateTab(tabConfig)
        if typeof(tabConfig) == "string" then tabConfig = {Name = tabConfig, Icon = "◆"} end
        tabConfig = tabConfig or {}
        tabConfig.Name = tabConfig.Name or "Tab"
        tabConfig.Icon = tabConfig.Icon or "◆"
        local pageIdx = #W._pages + 1

        -- Sidebar button (parented to sideStack, NOT sidebar: see sidebar note)
        local sbtn = Instance.new("TextButton")
        sbtn.Size = UDim2.new(1, 0, 0, 32)
        sbtn.BackgroundColor3 = Color3.fromRGB(16,16,21)
        sbtn.Text = ""; sbtn.BorderSizePixel = 0; sbtn.AutoButtonColor = false
        sbtn.ZIndex = 12; sbtn.Parent = sideStack
        sbtn.LayoutOrder = pageIdx
        Cr(sbtn, 5)
        local sicon = Instance.new("TextLabel")
        sicon.Size = UDim2.new(0, 30, 1, 0)
        sicon.BackgroundTransparency = 1
        sicon.Text = tabConfig.Icon
        sicon.TextColor3 = TXT_DIM
        sicon.Font = W.CurrentFont; sicon.TextSize = 14
        sicon.ZIndex = 13; sicon.Parent = sbtn
        table.insert(W._fonted, sicon)
        local slbl = Instance.new("TextLabel")
        slbl.Size = UDim2.new(1, -32, 1, 0)
        slbl.Position = UDim2.new(0, 30, 0, 0)
        slbl.BackgroundTransparency = 1
        slbl.Text = tabConfig.Name
        slbl.TextColor3 = TXT_DIM
        slbl.Font = W.CurrentFont; slbl.TextSize = 12
        slbl.TextXAlignment = Enum.TextXAlignment.Left
        slbl.TextTruncate = Enum.TextTruncate.AtEnd
        slbl.ZIndex = 13; slbl.Parent = sbtn
        table.insert(W._fonted, slbl)
        local sbar = Instance.new("Frame")
        sbar.Size = UDim2.new(0, 3, 0, 20)
        sbar.Position = UDim2.new(0, 0, 0.5, -10)
        sbar.BackgroundColor3 = W.CurrentAccent
        sbar.BorderSizePixel = 0; sbar.Visible = false
        W:_registerBreath(sbar, "BackgroundColor3")
        sbar.ZIndex = 13; sbar.Parent = sbtn
        Cr(sbar, 2)
        local sdot = Instance.new("Frame")
        sdot.Size = UDim2.new(0, 6, 0, 6)
        sdot.Position = UDim2.new(1, -12, 0, 6)
        sdot.BackgroundColor3 = W.CurrentAccent
        sdot.BorderSizePixel = 0; sdot.Visible = false
        sdot.ZIndex = 13; sdot.Parent = sbtn
        Cr(sdot, 3)
        local btnRec = {_frame = sbtn, _label = slbl, _icon = sicon, _bar = sbar, _active = false}
        table.insert(W._pageBtns, btnRec)
        sbtn.MouseButton1Click:Connect(function() W:showPage(pageIdx) end)
        sbtn.MouseEnter:Connect(function()
            if not btnRec._active then
                pcall(function()
                    TS:Create(sbtn, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                        {BackgroundColor3 = Color3.fromRGB(23,23,29)}):Play()
                end)
            end
        end)
        sbtn.MouseLeave:Connect(function()
            if not btnRec._active then
                pcall(function()
                    TS:Create(sbtn, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                        {BackgroundColor3 = Color3.fromRGB(16,16,21)}):Play()
                end)
            end
        end)
        -- Entrance: fade/slide each tab button in with a slight stagger so the
        -- sidebar doesn't just blink into existence on load.
        do
            local tBg = math.clamp(W.UITransparencyVal / 100, 0, 1)
            sbtn.BackgroundTransparency = 1
            slbl.TextTransparency = 1
            sicon.TextTransparency = 1
            local d = math.clamp((pageIdx - 1) * 0.06, 0, 0.3)
            task.delay(d, function()
                if sbtn and sbtn.Parent then
                    pcall(function()
                        TS:Create(sbtn, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                            {BackgroundTransparency = tBg}):Play()
                        TS:Create(slbl, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                            {TextTransparency = 0}):Play()
                        TS:Create(sicon, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                            {TextTransparency = 0}):Play()
                    end)
                    task.delay(0.25, function()
                        if sbtn and sbtn.Parent then
                            sbtn.BackgroundTransparency = tBg
                            slbl.TextTransparency = 0
                            sicon.TextTransparency = 0
                        end
                    end)
                end
            end)
        end

        -- Page (scrolling)
        local page = Instance.new("ScrollingFrame")
        page.Name = "Page_" .. tabConfig.Name
        page.Size = UDim2.new(1, 0, 1, 0)
        page.BackgroundTransparency = 1
        page.BorderSizePixel = 0
        page.ScrollBarThickness = 3
        page.ScrollBarImageColor3 = Color3.fromRGB(60,60,70)
        page.ZIndex = 11
        page.Visible = false
        page.Parent = content
        page.AutomaticCanvasSize = Enum.AutomaticSize.Y
        page.CanvasSize = UDim2.new(0, 0, 0, 0)
        local pad = Instance.new("UIPadding", page)
        pad.PaddingTop = UDim.new(0, 10); pad.PaddingBottom = UDim.new(0, 10)
        pad.PaddingLeft = UDim.new(0, 12); pad.PaddingRight = UDim.new(0, 12)
        local layout = Instance.new("UIListLayout", page)
        layout.SortOrder = Enum.SortOrder.LayoutOrder
        layout.Padding = UDim.new(0, 6)
        table.insert(W._pages, {frame = page, dot = sdot, name = tabConfig.Name,
            cards = {}, cardRows = {}, itemsSinceRow = 0})

        local T = {_page = page, _w = W, _pageIdx = pageIdx}
        T._list = page
        local orderCtr = 0
        local function nextOrder() orderCtr += 1 return orderCtr end
        local pageRec = W._pages[#W._pages]

        -- Card grid: 2 columns desktop, 1 column compact. Cards redistribute
        -- on demand without rebuilding their contents.
        local function relayoutCards()
            local perRow = W.Compact and 1 or 2
            for _, c in ipairs(pageRec.cards) do c.Parent = nil end
            for _, r in ipairs(pageRec.cardRows) do r:Destroy() end
            pageRec.cardRows = {}
            local i = 1
            while i <= #pageRec.cards do
                local row = Instance.new("Frame")
                row.Name = "CardRow"
                row.Size = UDim2.new(1, 0, 0, 0)
                row.AutomaticSize = Enum.AutomaticSize.Y
                row.BackgroundTransparency = 1
                row.BorderSizePixel = 0
                row.Parent = page
                local rowLayout = Instance.new("UIListLayout", row)
                rowLayout.FillDirection = Enum.FillDirection.Horizontal
                rowLayout.SortOrder = Enum.SortOrder.LayoutOrder
                rowLayout.Padding = UDim.new(0, 6)
                rowLayout.VerticalAlignment = Enum.VerticalAlignment.Top
                table.insert(pageRec.cardRows, row)
                for k = 1, perRow do
                    local card = pageRec.cards[i]
                    if not card then break end
                    card.Size = (perRow == 1) and UDim2.new(1, 0, 0, 0)
                        or UDim2.new(0.5, -3, 0, 0)
                    card.LayoutOrder = k
                    card.Parent = row
                    i += 1
                end
            end
            pageRec.itemsSinceRow = 0
        end
        pageRec.relayout = relayoutCards

        function T:CreateCard(name, icon)
            icon = icon or "▸"
            local last = pageRec.cardRows[#pageRec.cardRows]
            local perRow = W.Compact and 1 or 2
            if last then
                if #last:GetChildren() - 1 >= perRow then
                    last = nil
                else
                    local maxOrd = 0
                    for _, ch in ipairs(page:GetChildren()) do
                        if ch:IsA("GuiObject") then
                            maxOrd = math.max(maxOrd, ch.LayoutOrder or 0)
                        end
                    end
                    if last.LayoutOrder ~= maxOrd then last = nil end
                end
            end
            if not last then
                last = Instance.new("Frame")
                last.Name = "CardRow"
                last.Size = UDim2.new(1, 0, 0, 0)
                last.AutomaticSize = Enum.AutomaticSize.Y
                last.BackgroundTransparency = 1
                last.BorderSizePixel = 0
                last.Parent = page
                last.LayoutOrder = nextOrder()
                local rowLayout = Instance.new("UIListLayout", last)
                rowLayout.FillDirection = Enum.FillDirection.Horizontal
                rowLayout.SortOrder = Enum.SortOrder.LayoutOrder
                rowLayout.Padding = UDim.new(0, 6)
                rowLayout.VerticalAlignment = Enum.VerticalAlignment.Top
                table.insert(pageRec.cardRows, last)
                pageRec.itemsSinceRow = 0
            end
            local card = Instance.new("Frame")
            card.Name = "Card_" .. tostring(name)
            card.Size = (perRow == 1) and UDim2.new(1, 0, 0, 0)
                or UDim2.new(0.5, -3, 0, 0)
            card.AutomaticSize = Enum.AutomaticSize.Y
            card.BackgroundColor3 = CARD_BG
            card.BorderSizePixel = 0
            card.LayoutOrder = #last:GetChildren()
            card.Parent = last
            Cr(card, 6)
            local cardStroke = Instance.new("UIStroke", card)
            cardStroke.Color = BORDER
            cardStroke.Thickness = 1
            cardStroke.Transparency = 0.15
            W:_surface(card)
            -- Subtle card hover: edge lifts toward accent, then settles back.
            card.MouseEnter:Connect(function()
                pcall(function()
                    TS:Create(cardStroke, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                        {Color = W.CurrentAccent, Transparency = 0.45}):Play()
                end)
            end)
            card.MouseLeave:Connect(function()
                pcall(function()
                    TS:Create(cardStroke, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                        {Color = BORDER, Transparency = 0.15}):Play()
                end)
                cardStroke.Color = BORDER
                cardStroke.Transparency = 0.15
            end)
            local cardPad = Instance.new("UIPadding", card)
            cardPad.PaddingTop = UDim.new(0, 10); cardPad.PaddingBottom = UDim.new(0, 10)
            cardPad.PaddingLeft = UDim.new(0, 12); cardPad.PaddingRight = UDim.new(0, 12)
            local cardLayout = Instance.new("UIListLayout", card)
            cardLayout.SortOrder = Enum.SortOrder.LayoutOrder
            cardLayout.Padding = UDim.new(0, 6)
            local head = Instance.new("Frame")
            head.Size = UDim2.new(1, 0, 0, 20)
            head.BackgroundTransparency = 1
            head.LayoutOrder = -100
            head.Parent = card
            local hic = Instance.new("TextLabel")
            hic.Size = UDim2.new(0, 20, 1, 0)
            hic.BackgroundTransparency = 1
            hic.Text = icon
            hic.TextColor3 = W.CurrentAccent
            hic.Font = W.CurrentFont; hic.TextSize = 12
            hic.ZIndex = 12; hic.Parent = head
            table.insert(W._fonted, hic)
            table.insert(W._accentTexts, hic)
            local hname = Instance.new("TextLabel")
            hname.Size = UDim2.new(1, -22, 1, 0)
            hname.Position = UDim2.new(0, 22, 0, 0)
            hname.BackgroundTransparency = 1
            hname.Text = string.upper(tostring(name))
            hname.TextColor3 = W.CurrentAccent
            hname.Font = Enum.Font.GothamBold; hname.TextSize = 11
            hname.TextXAlignment = Enum.TextXAlignment.Left
            hname.TextTruncate = Enum.TextTruncate.AtEnd
            hname.ZIndex = 12; hname.Parent = head
            table.insert(W._accentTexts, hname)
            local hdiv = Instance.new("Frame")
            hdiv.Size = UDim2.new(1, 0, 0, 1)
            hdiv.BackgroundColor3 = BORDER
            hdiv.BorderSizePixel = 0
            hdiv.LayoutOrder = -99
            hdiv.Parent = card
            table.insert(pageRec.cards, card)
            local C = {_list = card, _w = W, _pageIdx = pageIdx}
            C.CreateSection = T.CreateSection
            C.CreateDivider = T.CreateDivider
            C.CreateToggle = T.CreateToggle
            C.CreateButton = T.CreateButton
            C.CreateSlider = T.CreateSlider
            C.CreateDropdown = T.CreateDropdown
            C.CreateLabel = T.CreateLabel
            C.CreateInput = T.CreateInput
            C.CreateParagraph = T.CreateParagraph
            W:_registerSearch(card, name, pageIdx)
            -- appear animation: subtle pop, no layout fight (scale only)
            local pop = Instance.new("UIScale", card)
            pop.Scale = 0.96
            TS:Create(pop, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                {Scale = 1}):Play()
            task.delay(0.2, function() pcall(function() pop:Destroy() end) end)
            return C
        end

        local function bindKeyMenu(Wref, row, name, getState, setState)
            row.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton2 then
                    local current = Wref.Keybinds[name]
                    Wref:ShowContextMenu({
                        {Header = true, Label = name},
                        {Label = "Enable", Callback = function() setState(true) end},
                        {Label = "Disable", Callback = function() setState(false) end},
                        {Separator = true},
                        {Label = "Bind Key" .. (current and (" [" .. current .. "]") or ""),
                            Color = Color3.fromRGB(200,180,100), Callback = function()
                            Wref._listening = {module = name}
                            Wref:Notify({Title = name, Content = "Press any key... (Esc to cancel)",
                                Type = "info", Duration = 5})
                        end},
                        {Label = "Clear Keybind", Callback = function()
                            Wref.Keybinds[name] = nil
                            Wref:Notify({Title = name, Content = "Keybind cleared", Type = "info", Duration = 2})
                        end},
                    }, input.Position)
                end
            end)
        end

        function T:CreateToggle(opts)
            local state = opts.Default or false
            local cb = opts.Callback
            local row = Instance.new("TextButton")
            row.Name = "Mod_" .. opts.Name
            row.Size = UDim2.new(1, 0, 0, rowHeight())
            row.BackgroundColor3 = state and EnabledBg(self._w.CurrentAccent) or ROW_BG
            row.Text = ""; row.BorderSizePixel = 0; row.AutoButtonColor = false
            row.ZIndex = 11; row.Parent = self._list
            row.LayoutOrder = nextOrder()
            Cr(row, 5)
            self._w:_surface(row)
            self._w:_registerSearch(row, opts.Name, self._pageIdx)
            local lbl = Instance.new("TextLabel")
            lbl.Size = UDim2.new(1, -52, 1, 0)
            lbl.Position = UDim2.new(0, 12, 0, 0)
            lbl.BackgroundTransparency = 1; lbl.Text = opts.Name
            lbl.TextColor3 = state and Lighten(self._w.CurrentAccent, 0.4) or TXT_DIM
            lbl.Font = self._w.CurrentFont; lbl.TextSize = 12
            lbl.TextXAlignment = Enum.TextXAlignment.Left
            lbl.TextTruncate = Enum.TextTruncate.AtEnd
            lbl.ZIndex = 12; lbl.Parent = row
            table.insert(self._w._fonted, lbl)
            local sw = Instance.new("Frame")
            sw.Name = "Switch"
            sw.Size = UDim2.new(0, 26, 0, 13)
            sw.Position = UDim2.new(1, -36, 0.5, -6.5)
            sw.BackgroundColor3 = state and self._w.CurrentAccent or Color3.fromRGB(45,45,50)
            sw.BorderSizePixel = 0; sw.ZIndex = 12; sw.Parent = row
            Cr(sw, 6)
            local kb = Instance.new("Frame")
            kb.Name = "Knob"
            kb.Size = UDim2.new(0, 11, 0, 11)
            kb.Position = state and UDim2.new(0, 14, 0.5, -5.5) or UDim2.new(0, 1, 0.5, -5.5)
            kb.BackgroundColor3 = state and Color3.new(1,1,1) or Color3.fromRGB(200,200,200)
            kb.BorderSizePixel = 0; kb.ZIndex = 13; kb.Parent = sw
            Cr(kb, 5)
            local function update(animate)
                local eb = EnabledBg(self._w.CurrentAccent)
                local tRow = state and eb or ROW_BG
                local tLbl = state and Lighten(self._w.CurrentAccent, 0.4) or TXT_DIM
                local tSw = state and self._w.CurrentAccent or Color3.fromRGB(45,45,50)
                local tKbPos = state and UDim2.new(0, 14, 0.5, -5.5) or UDim2.new(0, 1, 0.5, -5.5)
                local tKb = state and Color3.new(1,1,1) or Color3.fromRGB(200,200,200)
                if animate == false then
                    row.BackgroundColor3 = tRow
                    lbl.TextColor3 = tLbl
                    sw.BackgroundColor3 = tSw
                    kb.Position = tKbPos
                    kb.BackgroundColor3 = tKb
                else
                    pcall(function()
                        local ti = TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
                        TS:Create(row, ti, {BackgroundColor3 = tRow}):Play()
                        TS:Create(lbl, ti, {TextColor3 = tLbl}):Play()
                        TS:Create(sw, ti, {BackgroundColor3 = tSw}):Play()
                        TS:Create(kb, ti, {Position = tKbPos, BackgroundColor3 = tKb}):Play()
                    end)
                    row.BackgroundColor3 = tRow
                    lbl.TextColor3 = tLbl
                    sw.BackgroundColor3 = tSw
                    kb.Position = tKbPos
                    kb.BackgroundColor3 = tKb
                end
            end
            table.insert(self._w._updaters, update)
            local modObj = {Frame = row, Get = function() return state end,
                Set = function(_, v, silent)
                    state = (v == true)
                    update()
                    self._w.Modules[opts.Name] = modObj
                    self._w:_UpdateHUD()
                    if cb then task.spawn(cb, state) end
                    if not silent then
                        self._w:Notify({Title = opts.Name, Content = state and "enabled" or "disabled",
                            Type = state and "success" or "info", Duration = 2})
                    end
                end}
            row.MouseButton1Click:Connect(function()
                state = not state
                update()
                self._w.Modules[opts.Name] = modObj
                self._w:_UpdateHUD()
                if cb then task.spawn(cb, state) end
                self._w:Notify({Title = opts.Name, Content = state and "enabled" or "disabled",
                    Type = state and "success" or "info", Duration = 2})
            end)
            row.MouseEnter:Connect(function()
                if not state then
                    pcall(function()
                        TS:Create(row, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                            {BackgroundColor3 = ROW_BG_HOVER}):Play()
                    end)
                    row.BackgroundColor3 = ROW_BG_HOVER
                end
            end)
            row.MouseLeave:Connect(function()
                if not state then
                    pcall(function()
                        TS:Create(row, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                            {BackgroundColor3 = ROW_BG}):Play()
                    end)
                    row.BackgroundColor3 = ROW_BG
                end
            end)
            bindKeyMenu(self._w, row, opts.Name,
                function() return state end,
                function(v) modObj:Set(v) end)
            self._w.Modules[opts.Name] = modObj
            return modObj
        end

            Cr(handle, 7)
            table.insert(self._w._updaters, function()
                fill.BackgroundColor3 = self._w.CurrentAccent
                handle.BackgroundColor3 = self._w.CurrentAccent
                vlbl.TextColor3 = self._w.CurrentAccent
            end)
            local sliding = false
            local dsx, dsp = 0, pct
            local function apply(nv, silent)
                val = math.floor(math.clamp(nv, min, max) + 0.5)
                pct = (max > min) and ((val - min) / (max - min)) or 0
                fill.Size = UDim2.new(pct, 0, 1, 0)
                handle.Position = UDim2.new(pct, -7, 0.5, -7)
                vlbl.Text = tostring(val)
                -- silent = suppress notify spam, NOT state: config loads and
                -- programmatic Set(..., true) must still apply (FOV/scale/etc).
                if cb then task.spawn(cb, val) end
            end
            handle.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                    sliding = true; dsx = input.Position.X; dsp = pct
                end
            end)
            track.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                    local w = math.max(track.AbsoluteSize.X, 1)
                    local rel = math.clamp((input.Position.X - track.AbsolutePosition.X) / w, 0, 1)
                    apply(min + rel * (max - min))
                    sliding = true; dsx = input.Position.X; dsp = rel
                end
            end)
            local moveConn = UIS.InputChanged:Connect(function(input)
                if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch) then
                    local w = math.max(track.AbsoluteSize.X, 1)
                    apply(min + math.clamp(dsp + (input.Position.X - dsx) / w, 0, 1) * (max - min))
                end
            end)
            local endConn = UIS.InputEnded:Connect(function(input)
                if sliding and (input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch) then
                    sliding = false
                end
            end)
            table.insert(self._w._connections, moveConn)
            table.insert(self._w._connections, endConn)
            local so = {Frame = row, Get = function() return val end,
                Set = function(_, v, silent) apply(v, silent) end}
            self._w.Modules[opts.Name] = so
            return so
        end

    -- ============================================================
    -- NOTIFY
    -- ============================================================
    -- Centralized notification layout: one function owns every slot, so
    -- stacking, reflow, runtime position changes and resizes stay consistent.
    local NOTIF_GAP = 8
    function W:notifSlot(n)
        local pos = W.CurrentNotifPos
        if pos == "Top-Right" then
            return UDim2.new(1, -NOTIF_W - 10, 0, 12 + n * (NOTIF_H + NOTIF_GAP))
        elseif pos == "Top-Left" then
            return UDim2.new(0, 10, 0, 12 + n * (NOTIF_H + NOTIF_GAP))
        elseif pos == "Top-Center" then
            -- AnchorPoint is (0.5, 0) for centered, so X offset must be 0 (not -W/2).
            return UDim2.new(0.5, 0, 0, 12 + n * (NOTIF_H + NOTIF_GAP))
        elseif pos == "Bottom-Left" then
            return UDim2.new(0, 10, 1, -12 - (n + 1) * NOTIF_H - n * NOTIF_GAP)
        elseif pos == "Bottom-Center" then
            return UDim2.new(0.5, 0, 1, -12 - (n + 1) * NOTIF_H - n * NOTIF_GAP)
        else -- Bottom-Right
            return UDim2.new(1, -NOTIF_W - 10, 1, -12 - (n + 1) * NOTIF_H - n * NOTIF_GAP)
        end
    end

    local function notifIsTop()
        local pos = W.CurrentNotifPos
        return pos == "Top-Right" or pos == "Top-Left" or pos == "Top-Center"
    end

    function W:layoutNotifs(animate)
        local list = {}
        for _, f in pairs(W.ActiveNotifs) do
            if f and f.Parent then table.insert(list, f) end
        end
        table.sort(list, function(a, b)
            return (W._notifOrder[a] or 0) < (W._notifOrder[b] or 0)
        end)
        for i, fr in ipairs(list) do
            local target = W:notifSlot(i - 1)
            if animate == false then
                fr.Position = target
            else
                TS:Create(fr, TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                    {Position = target}):Play()
            end
        end
    end

    function W:Notify(opts)
        if typeof(opts) == "string" then opts = {Title = "NexaUI", Content = opts, Duration = 4} end
        opts.Title = opts.Title or "NexaUI"; opts.Content = opts.Content or ""
        opts.Duration = opts.Duration or 4; opts.Type = opts.Type or "info"
        pcall(function()
            local colors = W.NotifColors[opts.Type] or W.NotifColors.info
            local centered = W.CurrentNotifPos == "Top-Center" or W.CurrentNotifPos == "Bottom-Center"
            local fr = Instance.new("Frame"); fr.Size = UDim2.new(0,NOTIF_W,0,NOTIF_H)

            tt.Font = W.CurrentFont; tt.TextSize = 11; tt.TextXAlignment = Enum.TextXAlignment.Left
            tt.ZIndex = 901; tt.Parent = fr; table.insert(W._fonted, tt)
            local ct = Instance.new("TextLabel"); ct.Size = UDim2.new(1,-20,0,22); ct.Position = UDim2.new(0,14,0,20)
            ct.BackgroundTransparency = 1; ct.Text = opts.Content; ct.TextColor3 = Color3.fromRGB(160,160,170)
            ct.Font = W.CurrentFont; ct.TextSize = 9; ct.TextWrapped = true
            ct.TextXAlignment = Enum.TextXAlignment.Left; ct.ZIndex = 901; ct.Parent = fr
            table.insert(W._fonted, ct)
            -- Monotonic sequence (tick() can collide when spamming 3+ notifs).
            W._notifSeq += 1
            local slot = W._notifSeq
            W._notifOrder[fr] = slot
            -- count current actives for the entry slot, start pushed outward, then settle
            local n = 0
            for _ in pairs(W.ActiveNotifs) do n += 1 end
            local target = W:notifSlot(n)
            local push = (notifIsTop() and -1 or 1) * (NOTIF_H + 20)
            fr.Position = target + UDim2.new(0, 0, 0, push)
            W.ActiveNotifs[slot] = fr
            W:layoutNotifs(true)
            task.delay(opts.Duration, function()
                if W.ActiveNotifs[slot] then
                    local out = fr.Position + UDim2.new(0, 0, 0, (notifIsTop() and -1 or 1) * (NOTIF_H + 20))
                    TS:Create(fr, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.In),
                        {Position = out}):Play()
                    task.delay(0.28, function()
                        if fr then fr:Destroy() end
                        W.ActiveNotifs[slot] = nil
                        W._notifOrder[fr] = nil
                        W:layoutNotifs(true)
                    end)
                end
            end)
        end)
    end

    function W:SetConfigName(name)
        self._configName = name or "Default"
    end

    function W:SaveConfig(name)
        name = name or self._configName
        local ok, err = pcall(function()
            if typeof(isfolder) == "function" and typeof(makefolder) == "function" then
                if not isfolder(self.ConfigFolder) then makefolder(self.ConfigFolder) end
            end
            local cfg = {}
            for mn,mod in pairs(self.Modules) do
                if mod.Get then
                    local okg, v = pcall(function() return mod:Get() end)
                    if okg and typeof(v) ~= "Instance" and typeof(v) ~= "EnumItem" then
                        cfg[mn] = v
                    end
                end
            end
            cfg._AccentName = self.CurrentAccentName; cfg._FontName = self.CurrentFontName
            cfg._NotifPos = self.CurrentNotifPos; cfg._Breathing = self.BreathingEnabled
            cfg._UITransparency = self.UITransparencyVal; cfg._UIScale = self.UIScaleVal*100
            cfg._Keybinds = self.Keybinds
            local json = HS:JSONEncode(cfg)
            if typeof(writefile) == "function" then
                writefile(self.ConfigFolder.."/"..name..".json", json)
            else
                error("writefile unavailable (not an executor?)")
            end
        end)
        if ok then
            self:Notify({Title="Config",Content="Saved: "..name,Type="success",Duration=2})
        else
            self:Notify({Title="Config",Content="Save failed: "..tostring(err),Type="alert",Duration=3})
        end
    end

    function W:LoadConfig(name)
        name = name or self._configName
        local path = self.ConfigFolder.."/"..name..".json"
        local hasFile = false
        pcall(function()
            if typeof(isfile) == "function" then hasFile = isfile(path) end
        end)
        if not hasFile then self:Notify({Title="Config",Content="Not found: "..name,Type="warning",Duration=2}) return end
        local ok, data = pcall(function()
            if typeof(readfile) ~= "function" then error("readfile unavailable") end
            return HS:JSONDecode(readfile(path))
        end)
        if not ok or type(data) ~= "table" then self:Notify({Title="Config",Content="Failed to load",Type="alert",Duration=2}) return end
        -- Suppress theme Notify spam while still applying values + firing user
        -- callbacks (Toggle/Slider/Dropdown Set all fire cb even when silent).
        self._loading = true
        pcall(function()
            for mn,mod in pairs(self.Modules) do if data[mn] ~= nil and mod.Set then pcall(function() mod:Set(data[mn], true) end) end end
        end)
        self._loading = false
        if data._AccentName and AccentPresets[data._AccentName] then
            self.CurrentAccentName = data._AccentName; self.CurrentAccent = AccentPresets[self.CurrentAccentName]
            self:ApplyAccentColor(); if self.BreathingEnabled then self:StopBreathing() self:StartBreathing() end
        end
        if data._FontName and FontPresets[data._FontName] then self.CurrentFontName = data._FontName; self.CurrentFont = FontPresets[self.CurrentFontName]; self:ApplyFont() end
        if data._NotifPos then self.CurrentNotifPos = data._NotifPos self:layoutNotifs(true) end
        if data._Breathing ~= nil then self.BreathingEnabled = data._Breathing; if self.BreathingEnabled then self:StartBreathing() else self:StopBreathing() end end
        if data._UITransparency then self.UITransparencyVal = data._UITransparency; self:ApplyUITransparency(self.UITransparencyVal/100) end
        if data._UIScale then
            self.UIScaleVal = data._UIScale/100
            if self.WindowScale then self.WindowScale.Scale = self.UIScaleVal end
            if self.HUDScale then self.HUDScale.Scale = self.UIScaleVal end
        end
        -- FOV has no separate _FOV field; its slider already re-applied via Set cb above.
        if data._Keybinds and type(data._Keybinds) == "table" then self.Keybinds = data._Keybinds end
        self:Notify({Title="Config",Content="Loaded: "..name,Type="success",Duration=2})
    end

    function W:Destroy()
        self:StopBreathing()
        if W._toggleConn then pcall(function() W._toggleConn:Disconnect() end) end
        for _, conn in ipairs(W._connections) do pcall(function() conn:Disconnect() end) end
        W._connections = {}
        if W._camSizeConn then pcall(function() W._camSizeConn:Disconnect() end) end
        W._camSizeConn = nil
        for _,f in pairs(self.ActiveNotifs) do if f then pcall(function() f:Destroy() end) end end
        self.ActiveNotifs = {}
        self._notifOrder = {}
        if self.ScreenGui then pcall(function() self.ScreenGui:Destroy() end) end
        if NexaUI._lastWindow == self then
            NexaUI._lastWindow = nil
        end
    end

    W._toggleConn = UIS.InputBegan:Connect(function(input, gpe)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            -- Don't destroy the menu while its own buttons are being clicked.
            -- (UIS InputBegan fires before TextButton.Click and would eat the click.)
            local inside = false
            if W.ContextMenu.Visible then
                pcall(function()
                    inside = W:_pointInGui(W.ContextMenu, input.Position)
                end)
            end
            if not inside then
                W:HideContextMenu()
            end
        end
        if input.KeyCode == Enum.KeyCode.Escape and W._selector.openFor then
            W._closeSelector()
            return
        end
        if W._listening then
            if input.UserInputType == Enum.UserInputType.Keyboard then
                if input.KeyCode == Enum.KeyCode.Escape then
                    W:Notify({Title = W._listening.module, Content = "Bind cancelled", Type = "info", Duration = 2})
                else
                    W.Keybinds[W._listening.module] = input.KeyCode.Name
                    W:Notify({Title = W._listening.module, Content = "Bound to [" .. input.KeyCode.Name .. "]",
                        Type = "success", Duration = 2})
                end
                W._listening = nil
            end
            return
        end
        if input.KeyCode == W.KeybindToggle or input.KeyCode == Enum.KeyCode.Insert then
            W:ToggleUI()
            return
        end
        if gpe then return end
        for mn, kn in pairs(W.Keybinds) do
            if typeof(kn) == "string" then
                local ok, code = pcall(function() return Enum.KeyCode[kn] end)
                if ok and code and input.KeyCode == code then
                    local tm = W.Modules[mn]
                    if tm and tm.Get and tm.Set then
                        local okg, cur = pcall(function() return tm:Get() end)
                        if okg and typeof(cur) == "boolean" then
                            pcall(function() tm:Set(not cur, false) end)
                        end
                    end
                end
            end
        end
    end)

        lbl.TextColor3 = bs and Lighten(W.CurrentAccent, 0.4) or TXT_DIM
        lbl.Font = W.CurrentFont; lbl.TextSize = 12
        lbl.TextXAlignment = Enum.TextXAlignment.Left
        lbl.ZIndex = 12; lbl.Parent = row
        table.insert(W._fonted, lbl)
        local sw = Instance.new("Frame")
        sw.Size = UDim2.new(0, 26, 0, 13)
        sw.Position = UDim2.new(1, -36, 0.5, -6.5)
        sw.BackgroundColor3 = bs and W.CurrentAccent or Color3.fromRGB(45,45,50)
        sw.BorderSizePixel = 0; sw.ZIndex = 12; sw.Parent = row
        Cr(sw, 6)
        local function sync()
            row.BackgroundColor3 = bs and EnabledBg(W.CurrentAccent) or ROW_BG
            lbl.TextColor3 = bs and Lighten(W.CurrentAccent, 0.4) or TXT_DIM
            sw.BackgroundColor3 = bs and W.CurrentAccent or Color3.fromRGB(45,45,50)
        end
        table.insert(W._updaters, sync)
        local realSet = mod.Set
        mod.Set = function(_, v, silent)
            realSet(_, v)
            sync()
            W:_UpdateHUD()
            if not silent then
                W:Notify({Title = "Breathing", Content = bs and "enabled" or "disabled",
                    Type = bs and "success" or "info", Duration = 2})
            end
        end
        mod.Frame = row
        row.MouseButton1Click:Connect(function() mod:Set(not mod:Get()) W:_UpdateHUD() end)
    end
    ThemeTab:CreateDropdown({Name = "Accent", Options = AccentNames, Default = W.CurrentAccentName,
        Callback = function(v)
            if not AccentPresets[v] then return end
            W.CurrentAccentName = v
            W.CurrentAccent = AccentPresets[v]
            W:ApplyAccentColor()
            if W.BreathingEnabled then W:StopBreathing() W:StartBreathing() end
            if not W._loading then
                W:Notify({Title = "Theme", Content = "Accent: " .. v, Type = "info", Duration = 2})
            end
        end})
    ThemeTab:CreateDropdown({Name = "Font", Options = FontNames, Default = W.CurrentFontName,
        Callback = function(v)
            if not FontPresets[v] then return end
            W.CurrentFontName = v
            W.CurrentFont = FontPresets[v]
            W:ApplyFont()
            if not W._loading then
                W:Notify({Title = "Theme", Content = "Font: " .. v, Type = "info", Duration = 2})
            end
        end})
    ThemeTab:CreateDropdown({Name = "Notifications", Options = NotifPositions, Default = W.CurrentNotifPos,
        Callback = function(v)
            W.CurrentNotifPos = v
            W:layoutNotifs()
            if not W._loading then
                W:Notify({Title = "Theme", Content = "Position: " .. v, Type = "info", Duration = 2})
            end
        end})
    ThemeTab:CreateSection("DISPLAY")
    ThemeTab:CreateSlider({Name = "FOV", Min = 70, Max = 120, Default = 70,
        Callback = function(v)
            pcall(function()
                local cam = Workspace.CurrentCamera
                if cam then cam.FieldOfView = v end
            end)
        end})
    ThemeTab:CreateSlider({Name = "UI Transparency", Min = 0, Max = 70, Default = 0,
        Callback = function(v) W.UITransparencyVal = v W:ApplyUITransparency(v / 100) end})
    ThemeTab:CreateSlider({Name = "UI Scale", Min = 70, Max = 150, Default = 100,
        Callback = function(v)
            W.UIScaleVal = v / 100
            if W.WindowScale then W.WindowScale.Scale = W.UIScaleVal end
            if W.HUDScale then W.HUDScale.Scale = W.UIScaleVal end
        end})

    -- show first tab, fit layout, intro/load animation
    W:showPage(1)
    W:applyResponsive()
    -- Load animation: window pops in (scale 0.9 -> 1, Back easing) while the
    -- backdrop flashes and settles. Non-blocking; all targets snapped at the end
    -- so a failed tween can never leave the UI half-transparent.
    do
        local baseScale = W.UIScaleVal
        pcall(function()
            W.WindowScale.Scale = baseScale * 0.9
            TS:Create(W.WindowScale, TweenInfo.new(0.32, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
                {Scale = baseScale}):Play()
        end)
        W.Backdrop.Visible = true
        pcall(function()
            W.Backdrop.BackgroundTransparency = 1
            TS:Create(W.Backdrop, TweenInfo.new(0.32, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
                {BackgroundTransparency = 0.55}):Play()
        end)
        task.delay(0.38, function()
            if W.Backdrop and W.Backdrop.Parent then
                pcall(function()
                    TS:Create(W.Backdrop, TweenInfo.new(0.4), {BackgroundTransparency = 1}):Play()
                end)
                task.delay(0.45, function()
                    if W.Backdrop and W.Backdrop.Parent and W.UIOpen then
                        W.Backdrop.Visible = false
                    end
                end)
            end
            if W.WindowScale and W.WindowScale.Parent then
                W.WindowScale.Scale = W.UIScaleVal
            end
        end)
    end
    if W.BreathingEnabled then W:StartBreathing() end
    W:_UpdateHUD()
    W:Notify({Title = config.Name or "NexaUI", Content = "Loaded. Press " .. W.KeybindToggle.Name .. " to toggle.", Type = "success", Duration = 5})

    NexaUI._lastWindow = W
    return W
end
